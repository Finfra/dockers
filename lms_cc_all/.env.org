# === 모델 설정 ===
# lms 가 인식하는 모델 키(소문자 hub key). `lms get` 으로 받은 뒤 `lms ls` 가 보여주는 키를 사용.
#   - 검색 다운로드:  hub 키/검색어 또는 HF URL  (entrypoint 가 `lms get -y` 로 자동 시도)
#   - 예) "meta-llama-3.1-8b-instruct" (HF: lmstudio-community/Meta-Llama-3.1-8B-Instruct-GGUF)
# ⚠️ ANTHROPIC_MODEL 도 이 키와 동일해야 함(compose 가 LMS_MODEL 을 그대로 전달).
# air-gap: 외부망에서 사전 다운로드 후 LMS_MODEL_MOUNT 디렉토리에 배치하거나 이미지 내장.
LMS_MODEL=meta-llama-3.1-8b-instruct

# === LMS 백엔드 (headless lms CLI, OpenAI /v1) ===
LMS_PORT=1234
# 호스트 publish 바인드 주소: 127.0.0.1(기본) = 호스트 내부만 접근 가능.
#   Windows 등 다른 머신(LAN) 클라이언트가 접속하려면 0.0.0.0 으로 변경 (PATCH.md 참조).
LMS_BIND_HOST=127.0.0.1
# 서버 헬스 폴링 최대 횟수 (×2초). 대형 모델 초기화가 느리면 늘릴 것.
LMS_HEALTH_TRIES=30
# 컨텍스트 토큰 상한. Claude Code 는 시스템+도구 페이로드가 커서 8192 로는 부족 →
#   에이전트 용도면 32768 이상 권장. (단순 API 호출만이면 더 작아도 무방)
LMS_CONTEXT_LENGTH=32768
# GPU offload 비율: "max"(전체) / "off"(CPU) / 0~1 (부분). 기본 max.
LMS_GPU=max
# 동시 예측 슬롯 수. llmster 는 컨텍스트를 슬롯 수로 분할하므로(기본 4 → 슬롯당 ctx/4)
#   Claude Code 시스템 프롬프트(~16k)가 슬롯을 초과해 500 오류 발생 → 1 필수.
LMS_PARALLEL=1

# 모델 저장소 토글:
#   - 빈값: named volume(lms-models)에 격리 보관
#   - 호스트 경로: 기존 LM Studio 모델 디렉토리 공유 (ex: ~/.lmstudio/models)
# fg1: 호스트와 모델 공유 — 컨테이너/호스트 LM Studio 가 같은 저장소 사용 (중복 다운로드 방지)
LMS_MODEL_MOUNT=~/.lmstudio/models

# === Claude ↔ LMS ===
# 소형 GPU 추론 지연 대비 타임아웃(ms). 기본 10분.
API_TIMEOUT_MS=600000

# 호스트 코드 폴더 마운트 (선택) — docker-compose.code.yml override 사용 시
# MOUNT_CODE_DIR=~/code
MOUNT_CODE_DIR=

# === 컨테이너 (다중 인스턴스 동시 실행 시 충돌 회피) ===
COMPOSE_PROJECT_NAME=air_gap_claude_code_lms_one
LMS_CONTAINER_NAME=lms
CLAUDE_CONTAINER_NAME=claude
LMS_NETWORK_NAME=lms

# === 사용자 UID/GID (호스트 파일 권한 일치, 모델/코드 마운트 시 중요) ===
# Linux 호스트: `id -u`, `id -g` 결과로 맞출 것 권장
# macOS Docker Desktop: 1000 그대로 두어도 자동 위임됨
